# Org Semgrep policy fixture (GitHub Actions): deliberate findings, each next to a safe line.
# Stored as .yml.txt so it is never a workflow of this repository. tests/test_security_policy.py
# and .github/workflows/self-test.yml scan it as .github/workflows/workflow.yml; expected.txt
# lists what the rules in run-semgrep/rules/github-actions.yml must report on it.
name: fixture

on:
  pull_request:
  workflow_call:
    inputs:
      target:
        type: string
        default: "."

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7                                         # reported: a tag
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1  # silent: a full SHA
      - uses: actions/setup-python@main                                   # reported: a branch
      - uses: ./.github/actions/local                                     # silent: this checkout
      - uses: $/.github/actions/common/install-python                     # silent: this commit
      - uses: docker://alpine:3.22                                        # reported: no digest
      - uses: docker://alpine@sha256:0123456789abcdef0123456789abcdef0123456789abcdef0123456789abcdef

      - name: Title in the script
        run: echo "Checking ${{ github.event.pull_request.title }}"       # reported

      - name: Branch name and input in a multi-line script
        run: |                                                            # reported
          set -euo pipefail
          git switch "${{ github.head_ref }}"
          ls "${{ inputs.target }}"

      - name: The same values through env
        env:
          TITLE: ${{ github.event.pull_request.title }}
          TARGET: ${{ inputs.target }}
        run: |                                                            # silent
          echo "Checking $TITLE"
          ls "$TARGET"

      - name: A context the event author does not control
        run: echo "Run ${{ github.run_id }} on ${{ runner.os }}"          # silent

  call:
    uses: some-org/some-repo/.github/workflows/build.yml@v1               # reported: a tag
